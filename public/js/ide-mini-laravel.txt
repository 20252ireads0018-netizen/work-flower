        public static function avisar(string $chave, string $msg): void

            if (!isset(self::$avisados[$chave])) { self::$avisados[$chave] = 1; self::log('warn', $msg); }

        public static function relativo(string $arq): string
        {
            $arq = str_replace('\\', '/', $arq);
            $base = str_replace('\\', '/', self::$base);
            if (preg_match('#__views/([\w.\-]+)\.php$#', $arq, $m)) return 'view ' . $m[1];
            return str_starts_with($arq, $base) ? substr($arq, strlen($base)) : basename($arq);
        }
    }

    /** Objeto que aceita qualquer chamada e não faz nada (Mail, Queue, Event… não existem no navegador). */
    class Nada implements \ArrayAccess, \IteratorAggregate, \Countable
    {
        public function __call($m, $a) { return $this; }
        public static function __callStatic($m, $a) { return new static(); }
        public function __get($k) { return null; }
        public function __isset($k) { return false; }
        public function __toString(): string { return ''; }
        public function offsetExists(mixed $o): bool { return false; }
        public function offsetGet(mixed $o): mixed { return null; }
        public function offsetSet(mixed $o, mixed $v): void {}
        public function offsetUnset(mixed $o): void {}
        public function getIterator(): \Traversable { return new \ArrayIterator([]); }
        public function count(): int { return 0; }
    }
    class FachadaNada
    {
        public static function __callStatic($m, $a)
        {
            K::avisar(static::class, static::class . '::' . $m . '() não existe no Laravel do navegador e foi ignorado.');
            return new Nada();
        }
    }

    function padroes(string $arq): array
    {
        return match ($arq) {
            'app' => ['name' => env('APP_NAME', 'Laravel'), 'env' => env('APP_ENV', 'local'), 'debug' => true, 'url' => 'http://localhost',
                'timezone' => 'America/Sao_Paulo', 'locale' => 'pt_BR', 'fallback_locale' => 'en'],
            'database' => ['default' => 'sqlite'],
            'session' => ['driver' => 'array', 'lifetime' => 120],
            'auth' => ['defaults' => ['guard' => 'web']],
            default => [],
        };
    }

    function env_ler(): array
    {
        if (K::$env !== null) return K::$env;
        K::$env = [];
        $arq = K::caminho('.env');
        if (!is_file($arq)) return K::$env;
        foreach (file($arq, FILE_IGNORE_NEW_LINES) as $l) {
            if (!preg_match('/^\s*(?:export\s+)?([A-Za-z_][A-Za-z0-9_.]*)\s*=\s*(.*)$/', $l, $m)) continue;
            $v = trim($m[2]);
            if ($v !== '' && ($v[0] === '"' || $v[0] === "'")) {
                $q = $v[0];
                $fim = strpos($v, $q, 1);
                $v = $fim === false ? substr($v, 1) : substr($v, 1, $fim - 1);
            } else {
                $v = trim(preg_replace('/\s+#.*$/', '', $v));
            }
            K::$env[$m[1]] = $v;
        }
        return K::$env;
    }

    function config_get(string $chave, $padrao = null)
    {
        $arq = explode('.', $chave)[0];
        if (!array_key_exists($arq, K::$config)) {
            $f = K::caminho('config/' . $arq . '.php');
            $v = [];
            if (is_file($f)) {
                $v = (static function ($__f) { return require $__f; })($f);
                if (!is_array($v)) $v = [];
            }
            K::$config[$arq] = array_replace_recursive(padroes($arq), $v);
        }
        return \Illuminate\Support\Arr::get(K::$config, $chave, $padrao);
    }

    /* ---------- banco (SQLite) ---------- */
    function pdo(): \PDO
    {
        if (!K::$pdo) {
            K::$pdo = new \PDO('sqlite:' . (K::$req['db'] ?? ':memory:'));
            K::$pdo->setAttribute(\PDO::ATTR_ERRMODE, \PDO::ERRMODE_EXCEPTION);
            K::$pdo->setAttribute(\PDO::ATTR_DEFAULT_FETCH_MODE, \PDO::FETCH_ASSOC);
            K::$pdo->exec('PRAGMA foreign_keys = ON');
        }
        return K::$pdo;
    }
    function valorSql($v)
    {
        if (is_bool($v)) return (int) $v;
        if ($v instanceof \DateTimeInterface) return $v->format('Y-m-d H:i:s');
        if ($v instanceof \BackedEnum) return $v->value;
        if (is_array($v)) return json_encode($v, JSON_UNESCAPED_UNICODE);
        if ($v instanceof \Illuminate\Database\Eloquent\Model) return $v->getKey();
        if (is_object($v) && method_exists($v, '__toString')) return (string) $v;
        return $v;
    }
    function q(string $sql, array $b = []): \PDOStatement
    {
        K::$meta['consultas'] = (K::$meta['consultas'] ?? 0) + 1;
        try {
            $st = pdo()->prepare($sql);
            $st->execute(array_map('WF\valorSql', array_values($b)));
            return $st;
        } catch (\PDOException $e) {
            $sqlComValores = $sql;
            foreach ($b as $v) {
                $v = valorSql($v);
                $sqlComValores = preg_replace('/\?/', is_null($v) ? 'null' : (is_numeric($v) ? (string) $v : "'" . str_replace("'", "''", (string) $v) . "'"), $sqlComValores, 1);
            }
            throw new \Illuminate\Database\QueryException($e->getMessage() . ' (Connection: sqlite, SQL: ' . $sqlComValores . ')', 0, $e);
        }
    }
    function tabelaExiste(string $t): bool
    {
        return (bool) q("select count(*) from sqlite_master where type = 'table' and name = ?", [$t])->fetchColumn();
    }
    function aspas(string $id): string
    {
        if ($id === '*') return '*';
        if (preg_match('/^(.+)\s+as\s+(\w+)$/i', $id, $m)) return aspas(trim($m[1])) . ' as "' . $m[2] . '"';
        if (str_contains($id, '(') || str_contains($id, ' ')) return $id;
        return implode('.', array_map(fn ($p) => $p === '*' ? '*' : '"' . str_replace('"', '""', $p) . '"', explode('.', $id)));
    }

    /* ---------- migrations e seeders ---------- */
    function migrationsArqs(): array
    {
        $d = K::caminho('database/migrations');
        $l = is_dir($d) ? glob($d . '/*.php') : [];
        sort($l);
        return $l;
    }
    function carregarMigration(string $arq)
    {
        $antes = get_declared_classes();
        $m = (static function ($__f) { return require $__f; })($arq);
        if (is_object($m)) return $m;
        foreach (array_diff(get_declared_classes(), $antes) as $c) if (is_subclass_of($c, \Illuminate\Database\Migrations\Migration::class)) return new $c();
        return null;
    }
    function garantirMigrations(): bool
    {
        if (tabelaExiste('migrations')) return false;
        pdo()->exec('create table "migrations" ("id" integer primary key autoincrement not null, "migration" varchar not null, "batch" integer not null)');
        return true;
    }
    function migrar(): array
    {
        garantirMigrations();
        $feitas = q('select migration from migrations')->fetchAll(\PDO::FETCH_COLUMN);
        $lote = (int) q('select max(batch) from migrations')->fetchColumn() + 1;
        $rodou = [];
        foreach (migrationsArqs() as $arq) {
            $nome = basename($arq, '.php');
            if (in_array($nome, $feitas, true)) continue;
            $m = carregarMigration($arq);
            if ($m && method_exists($m, 'up')) $m->up();
            q('insert into migrations (migration, batch) values (?, ?)', [$nome, $lote]);
            $rodou[] = $nome;
        }
        return $rodou;
    }
    function reverter(int $passos = 1): array
    {
        if (!tabelaExiste('migrations')) return [];
        $feitas = [];
        for ($i = 0; $i < $passos; $i++) {
            $lote = (int) q('select max(batch) from migrations')->fetchColumn();
            if (!$lote) break;
            $nomes = q('select migration from migrations where batch = ? order by id desc', [$lote])->fetchAll(\PDO::FETCH_COLUMN);
            foreach ($nomes as $nome) {
                $arq = K::caminho('database/migrations/' . $nome . '.php');
                $m = is_file($arq) ? carregarMigration($arq) : null;
                if ($m && method_exists($m, 'down')) $m->down();
                q('delete from migrations where migration = ?', [$nome]);
                $feitas[] = $nome;
            }
        }
        return $feitas;
    }
    function apagarTudo(): void
    {
        pdo()->exec('PRAGMA foreign_keys = OFF');
        foreach (q("select name from sqlite_master where type = 'table' and name not like 'sqlite_%'")->fetchAll(\PDO::FETCH_COLUMN) as $t) pdo()->exec('drop table "' . $t . '"');
        pdo()->exec('PRAGMA foreign_keys = ON');
    }
    function semear(string $classe = 'Database\\Seeders\\DatabaseSeeder'): bool
    {
        if (!class_exists($classe)) return false;
        $s = make($classe);
        chamar([$s, 'run'], []);
        return true;
    }

    function autoload(string $c): void
    {
        $mapa = K::$psr4 + ['App\\' => 'app/', 'Database\\Seeders\\' => 'database/seeders/', 'Database\\Factories\\' => 'database/factories/', 'Database\\' => 'database/'];
        foreach ($mapa as $pre => $dir) {
            if (!str_starts_with($c, $pre)) continue;
            $f = K::caminho(rtrim($dir, '/') . '/' . str_replace('\\', '/', substr($c, strlen($pre))) . '.php');
            if (is_file($f)) { require_once $f; if (class_exists($c, false) || interface_exists($c, false) || trait_exists($c, false) || enum_exists($c, false)) return; }
        }
        stub($c);
    }
    function stub(string $c): void
    {
        $p = strrpos($c, '\\');
        $ns = $p === false ? '' : substr($c, 0, $p);
        $nome = $p === false ? $c : substr($c, $p + 1);
        $cab = $ns ? 'namespace ' . $ns . '; ' : '';
        if ($c === 'App\\Http\\Controllers\\Controller') { eval($cab . 'abstract class Controller extends \\Illuminate\\Routing\\Controller {}'); return; }
        if (!preg_match('/^(Illuminate|Laravel|Livewire|Inertia|Spatie|Barryvdh|Symfony|Carbon|Psr|Tests|App\\\\Providers|App\\\\Http\\\\Middleware)\\\\/', $c)) return;
        if (in_array($nome, TRAITS, true)) { eval($cab . 'trait ' . $nome . ' {}'); return; }
        if (str_contains($c, '\\Contracts\\') || str_ends_with($nome, 'Interface') || in_array($nome, ['MustVerifyEmail', 'ShouldQueue', 'Htmlable', 'Arrayable', 'Jsonable', 'Responsable', 'CastsAttributes'], true)) { eval($cab . 'interface ' . $nome . ' {}'); return; }
        if (str_contains($c, '\\Facades\\')) { eval($cab . 'class ' . $nome . ' extends \\WF\\FachadaNada {}'); return; }
        eval($cab . 'class ' . $nome . ' extends \\WF\\Nada {}');
        K::avisar($c, 'A classe ' . $c . ' não existe no Laravel do navegador: as chamadas a ela não fazem nada.');
    }

    /* ---------- contêiner e chamadas com injeção ---------- */
    function make(string $c, array $params = [])
    {
        if (isset(K::$singletons[$c])) return K::$singletons[$c];
        if ($c === \Illuminate\Http\Request::class) return K::$request;
        if (is_subclass_of($c, \Illuminate\Foundation\Http\FormRequest::class)) return formRequest($c);
        if ($c === \Illuminate\Session\Store::class) return K::$sessao;
        $rc = new \ReflectionClass($c);
        if (!$rc->isInstantiable()) throw new \Illuminate\Contracts\Container\BindingResolutionException('Target [' . $c . '] is not instantiable (é interface ou classe abstrata).');
        $ctor = $rc->getConstructor();
        return $ctor ? $rc->newInstanceArgs(argumentos($ctor, $params)) : new $c();
    }
    function argumentos(\ReflectionFunctionAbstract $f, array $params): array
    {
        $out = [];
        $usados = [];
        $proximo = function () use (&$params, &$usados) {
            foreach ($params as $k => $v) if (!isset($usados[$k])) { $usados[$k] = 1; return [true, $v]; }
            return [false, null];
        };
        foreach ($f->getParameters() as $p) {
            $nome = $p->getName();
            $t = $p->getType();
            $classe = ($t instanceof \ReflectionNamedType && !$t->isBuiltin()) ? $t->getName() : null;
            if ($p->isVariadic()) break;
            if ($classe && is_a($classe, \Illuminate\Http\Request::class, true)) {
                $out[] = $classe === \Illuminate\Http\Request::class ? K::$request : make($classe);
                continue;
            }
            if ($classe && is_subclass_of($classe, \Illuminate\Database\Eloquent\Model::class)) {
                if (array_key_exists($nome, $params)) { $usados[$nome] = 1; [$tem, $v] = [true, $params[$nome]]; } else [$tem, $v] = $proximo();
                if ($v instanceof $classe) { $out[] = $v; continue; }
                if (!$tem || $v === null) { $out[] = $p->isDefaultValueAvailable() ? $p->getDefaultValue() : new $classe(); continue; }
                $m = (new $classe())->resolveRouteBinding($v);
                if (!$m) throw (new \Illuminate\Database\Eloquent\ModelNotFoundException())->setModel($classe, [$v]);
                $out[] = $m;
                continue;
            }
            if ($classe && enum_exists($classe) && (array_key_exists($nome, $params))) { $usados[$nome] = 1; $out[] = $classe::from($params[$nome]); continue; }
            if ($classe && $classe !== 'Closure' && (class_exists($classe) || interface_exists($classe))) { $out[] = make($classe); continue; }
            if (array_key_exists($nome, $params)) { $usados[$nome] = 1; $out[] = $params[$nome]; continue; }
            if (!$p->isDefaultValueAvailable() && !$classe) { [$tem, $v] = $proximo(); if ($tem) { $out[] = $v; continue; } }
            if ($p->isDefaultValueAvailable()) $out[] = $p->getDefaultValue();
            elseif ($p->allowsNull()) $out[] = null;
            else throw new \ArgumentCountError('Não sei o que passar para o parâmetro $' . $nome . ' de ' . ($f instanceof \ReflectionMethod ? $f->class . '::' . $f->getName() . '()' : 'uma função') . '.');
        }
        return $out;
    }
    function chamar($acao, array $params)
    {
        if (is_array($acao) && is_string($acao[0])) $acao = [make($acao[0]), $acao[1]];
        if (is_array($acao)) {
            if (!method_exists($acao[0], $acao[1])) {
                if (method_exists($acao[0], '__call')) return $acao[0]->{$acao[1]}(...array_values($params));
                throw new \BadMethodCallException('Method ' . get_class($acao[0]) . '::' . $acao[1] . ' does not exist.');
            }
            $rf = new \ReflectionMethod($acao[0], $acao[1]);
            return $rf->invokeArgs($rf->isStatic() ? null : $acao[0], argumentos($rf, $params));
        }
        if (is_string($acao) && class_exists($acao)) return chamar([make($acao), '__invoke'], $params);
        if (is_object($acao) && !($acao instanceof \Closure) && method_exists($acao, '__invoke')) return chamar([$acao, '__invoke'], $params);
        $rf = new \ReflectionFunction($acao);
        return $rf->invokeArgs(argumentos($rf, $params));
    }
    function formRequest(string $classe)
    {
        $r = new $classe();
        $r->copiarDe(K::$request);
        if (method_exists($r, 'authorize') && $r->authorize() === false) throw new HttpException(403, 'This action is unauthorized.');
        if (method_exists($r, 'prepareForValidation')) $r->prepareForValidation();
        $v = \Illuminate\Support\Facades\Validator::make($r->all(), method_exists($r, 'rules') ? chamar([$r, 'rules'], []) : [],
            method_exists($r, 'messages') ? $r->messages() : [], method_exists($r, 'attributes') ? $r->attributes() : []);
        $r->validador = $v;
        $v->validate();
        if (method_exists($r, 'passedValidation')) $r->passedValidation();
        return $r;
    }

    /* ---------- respostas ---------- */
    function paraResposta($r): \Illuminate\Http\Response
    {
        if ($r instanceof \Illuminate\Http\Response) return $r;
        if ($r instanceof \Illuminate\Contracts\Support\Responsable) return paraResposta($r->toResponse(K::$request));
        if ($r instanceof \Illuminate\View\View) return new \Illuminate\Http\Response($r->render());
        if ($r instanceof \Illuminate\Contracts\Support\Htmlable) return new \Illuminate\Http\Response($r->toHtml());
        if (is_array($r) || $r instanceof \JsonSerializable || $r instanceof \Illuminate\Contracts\Support\Arrayable) return new \Illuminate\Http\JsonResponse($r);
        if ($r instanceof \Stringable) return new \Illuminate\Http\Response((string) $r);
        if (is_bool($r)) return new \Illuminate\Http\Response($r ? '1' : '');
        if (is_object($r)) return new \Illuminate\Http\JsonResponse($r);
        return new \Illuminate\Http\Response((string) $r);
    }

    function paginaDeErro(\Throwable $e): \Illuminate\Http\Response
    {
        if ($e instanceof RespostaLancada) return paraResposta($e->resposta);
        if ($e instanceof \Illuminate\Validation\ValidationException) {
            if (K::$request && K::$request->expectsJson()) return new \Illuminate\Http\JsonResponse(['message' => $e->getMessage(), 'errors' => $e->errors()], 422);
            $r = redirect()->to($e->redirectTo ?? previousUrl());
            $r->withInput(K::$request ? K::$request->except(['password', 'password_confirmation', '_token', '_method']) : []);
            $r->withErrors($e->errors());
            return $r;
        }
        if ($e instanceof \Illuminate\Auth\AuthenticationException) {
            if (K::$request && K::$request->expectsJson()) return new \Illuminate\Http\JsonResponse(['message' => 'Unauthenticated.'], 401);
            if (\Illuminate\Support\Facades\Route::has('login')) return redirect()->guest(route('login'));
            $e = new HttpException(401, 'Unauthenticated.');
        }
        if ($e instanceof \Illuminate\Database\Eloquent\ModelNotFoundException) $e = new HttpException(404, $e->getMessage());
        if ($e instanceof HttpException) {
            $titulos = [401 => 'Não autenticado', 403 => 'Acesso negado', 404 => 'Não encontrado', 405 => 'Método não permitido', 419 => 'Página expirada', 422 => 'Dados inválidos', 429 => 'Muitas tentativas', 500 => 'Erro no servidor', 503 => 'Indisponível'];
            if ($e->status >= 400) K::log($e->status >= 500 ? 'error' : 'warn', $e->status . ' ' . ($titulos[$e->status] ?? '') . ($e->getMessage() ? ': ' . $e->getMessage() : '') . ' · ' . (K::$request ? K::$request->method() . ' /' . K::$request->path() : ''));
            if (K::$request && K::$request->expectsJson()) return new \Illuminate\Http\JsonResponse(['message' => $e->getMessage() ?: ($titulos[$e->status] ?? 'Erro')], $e->status, $e->headers);
            $v = 'errors.' . $e->status;
            if (\Illuminate\Support\Facades\View::exists($v)) return new \Illuminate\Http\Response(view($v, ['exception' => $e])->render(), $e->status);
            $html = '<!doctype html><meta charset="utf-8"><title>' . $e->status . '</title><style>body{margin:0;min-height:100vh;display:grid;place-items:center;font-family:system-ui,sans-serif;background:#f7f7f8;color:#555}div{font-size:1.15rem;letter-spacing:.03em}b{padding-right:1rem;margin-right:1rem;border-right:1px solid #ccc}p{text-align:center;color:#999;font-size:.85rem}</style><main><div><b>'
                . $e->status . '</b>' . htmlspecialchars($titulos[$e->status] ?? 'Erro') . '</div>' . ($e->getMessage() ? '<p>' . htmlspecialchars($e->getMessage()) . '</p>' : '') . '</main>';
            return new \Illuminate\Http\Response($html, $e->status, $e->headers);
        }
        // erro de programação: página no estilo do Laravel com o trecho do código
        [$arq, $linha] = localDoErro($e);
        K::$meta['erro'] = ['classe' => get_class($e), 'msg' => $e->getMessage(), 'arq' => $arq, 'linha' => $linha];
        K::log('error', get_class($e) . ': ' . $e->getMessage() . ' (' . $arq . ($linha ? ':' . $linha : '') . ')');
        if (K::$request && K::$request->expectsJson()) return new \Illuminate\Http\JsonResponse(['message' => $e->getMessage(), 'exception' => get_class($e), 'file' => $arq, 'line' => $linha], 500);
        return new \Illuminate\Http\Response(htmlErro($e, $arq, $linha), 500);
    }
    function localDoErro(\Throwable $e): array
    {
        // o primeiro quadro que é do projeto (não deste arquivo)
        $quadros = array_merge([['file' => $e->getFile(), 'line' => $e->getLine()]], $e->getTrace());
        foreach ($quadros as $q) {
            if (empty($q['file']) || $q['file'] === __FILE__ || str_contains($q['file'], 'eval()')) continue;
            return [K::relativo($q['file']), $q['line'] ?? 0, $q['file']];
        }
        return [K::relativo($e->getFile()), $e->getLine(), $e->getFile()];
    }
    function htmlErro(\Throwable $e, string $arq, int $linha): string
    {
        $h = fn ($s) => htmlspecialchars((string) $s, ENT_QUOTES, 'UTF-8');
        $trecho = '';
        $real = localDoErro($e)[2] ?? '';
        if ($real && is_file($real) && $linha) {
            $ls = file($real, FILE_IGNORE_NEW_LINES);
            for ($i = max(1, $linha - 6); $i <= min(count($ls), $linha + 4); $i++) {

            }
        }
        $pilha = '';
        foreach ($e->getTrace() as $t) {
            if (empty($t['file']) || $t['file'] === __FILE__) continue;
            $classe = preg_replace('/@anonymous.*$/s', '@anonymous', $t['class'] ?? '');
            $pilha .= '<li>' . $h(K::relativo($t['file'])) . ':' . ($t['line'] ?? '') . ' <span>' . $h($classe . ($t['type'] ?? '') . ($t['function'] ?? '')) . '()</span></li>';
            if (substr_count($pilha, '<li>') >= 8) break;
        }
        $req = K::$request ? $h(K::$request->method() . ' ' . K::$request->fullUrl()) : '';
        return '<!doctype html><html lang="pt-BR"><meta charset="utf-8"><title>' . $h(class_basename($e)) . '</title><style>'
            . 'body{margin:0;font:14px/1.5 system-ui,sans-serif;background:#f3f4f6;color:#111827}main{max-width:960px;margin:0 auto;padding:1.5rem}'

            . '.t{color:#ef4444;font-weight:600;font-size:.8rem;letter-spacing:.04em;text-transform:uppercase}h1{font-size:1.25rem;margin:.3rem 0 .6rem;word-break:break-word}'
            . '.a{color:#6b7280;font-family:ui-monospace,monospace;font-size:.85rem}pre{margin:0;overflow:auto;font:12.5px/1.6 ui-monospace,monospace;background:#111827;color:#e5e7eb;border-radius:.5rem;padding:.6rem 0}'
            . '.l{padding:0 .8rem;white-space:pre}.l i{display:inline-block;width:2.6rem;color:#6b7280;font-style:normal}.x{background:#7f1d1d}ol{margin:0;padding-left:1.2rem;font-family:ui-monospace,monospace;font-size:.82rem}ol span{color:#6b7280}'
            . '</style><main><div class="c"><div class="t">' . $h(get_class($e)) . '</div><h1>' . $h($e->getMessage()) . '</h1><div class="a">' . $h($arq) . ($linha ? ':' . $linha : '') . ' · ' . $req . '</div></div>'
            . ($trecho ? '<div class="c"><pre>' . $trecho . '</pre></div>' : '') . ($pilha ? '<div class="c"><b>Pilha de chamadas</b><ol>' . $pilha . '</ol></div>' : '')
            . '<div class="c a">Laravel do navegador (Work Flower) · PHP ' . PHP_VERSION . '</div></main></html>';
    }

    /** Arquivos de public/ (css, js) entram na página: a prévia não tem servidor para buscá-los. */
    function embutirAssets(string $html): string
    {
        $local = function (string $href): ?string {
            $u = preg_replace('#^https?://localhost(:\d+)?#i', '', $href);
            if (preg_match('#^(?:[a-z]+:)?//#i', $u) || str_starts_with($u, 'data:')) return null;
            $u = explode('?', explode('#', $u)[0])[0];
            $f = K::caminho('public/' . ltrim($u, '/'));
            return is_file($f) ? $f : null;
        };
        $html = preg_replace_callback('#<link\b[^>]*?href=["\']([^"\']+)["\'][^>]*>#i', function ($m) use ($local) {
            if (!preg_match('/stylesheet/i', $m[0]) || !($f = $local($m[1]))) return $m[0];
            return '<style>' . str_ireplace('</style', '<\/style', file_get_contents($f)) . '</style>';
        }, $html);
        return preg_replace_callback('#<script\b([^>]*?)\ssrc=["\']([^"\']+)["\']([^>]*)>\s*</script>#i', function ($m) use ($local) {
            if (!($f = $local($m[2]))) return $m[0];
            return '<script' . $m[1] . $m[3] . '>' . str_ireplace('</script', '<\/script', file_get_contents($f)) . '</script>';
        }, $html);
    }

    /* ---------- ciclo de uma requisição ---------- */
    function iniciar(array $req): void
    {
        K::$req = $req;
        if (!empty($req['base'])) K::$base = rtrim($req['base'], '/\\') . '/';
        date_default_timezone_set('America/Sao_Paulo');
        $cj = K::caminho('composer.json');
        if (is_file($cj)) {
            $j = json_decode((string) file_get_contents($cj), true);
            foreach (($j['autoload']['psr-4'] ?? []) as $ns => $dir) if (is_string($dir)) K::$psr4[$ns] = $dir;
            foreach (($j['autoload']['files'] ?? []) as $f) if (is_file(K::caminho($f))) $GLOBALS['__wf_arquivos'][] = K::caminho($f);
        }
        spl_autoload_register('WF\autoload');
        set_error_handler(function ($n, $msg, $arq, $linha) {
            if (!(error_reporting() & $n) || in_array($n, [E_DEPRECATED, E_USER_DEPRECATED], true)) return false;
            throw new \ErrorException($msg, 0, $n, $arq, $linha);
        });
        foreach ($GLOBALS['__wf_arquivos'] ?? [] as $f) require_once $f;
        K::$sessao = new \Illuminate\Session\Store($req['sessao'] ?? []);
        if (class_exists('App\\Models\\User')) K::$usuarioClasse = 'App\\Models\\User';
    }
    function carregarRotas(): void
    {
        $web = K::caminho('routes/web.php');
        $api = K::caminho('routes/api.php');
        if (is_file($web)) (new \Illuminate\Routing\RouteRegistrar())->middleware('web')->group($web);
        if (is_file($api)) (new \Illuminate\Routing\RouteRegistrar())->prefix('api')->middleware('api')->group($api);
    }
    function providers(): void
    {
        foreach (glob(K::caminho('app/Providers') . '/*.php') ?: [] as $f) {
            $c = 'App\\Providers\\' . basename($f, '.php');
            if (!class_exists($c) || !is_subclass_of($c, \Illuminate\Support\ServiceProvider::class)) continue;
            $p = new $c(null);
            if (method_exists($p, 'register')) $p->register();
            if (method_exists($p, 'boot')) chamar([$p, 'boot'], []);
        }
    }
    function bancoAutomatico(): void
    {
        if (!migrationsArqs()) return;
        $novo = garantirMigrations();
        $rodou = migrar();

    }

    function despachar(\Illuminate\Http\Request $req): \Illuminate\Http\Response
    {
        $metodo = $req->method();
        $caminho = '/' . trim($req->path(), '/');
        $permitidos = [];
        foreach (K::$rotas as $r) {
            $p = $r->casar($caminho);
            if ($p === null) continue;
            if (!in_array($metodo, $r->metodos, true)) { $permitidos = array_merge($permitidos, $r->metodos); continue; }
            return rodarRota($r, $p, $req);
        }
        if ($permitidos) throw new HttpException(405, 'The ' . $metodo . ' method is not supported for route ' . $caminho . '. Supported methods: ' . implode(', ', array_unique(array_diff($permitidos, ['HEAD']))) . '.');
        if (K::$fallback) return paraResposta(chamar(K::$fallback, []));
        throw new HttpException(404, 'Nenhuma rota para ' . $metodo . ' ' . $caminho . '. Confira routes/web.php (artisan route:list mostra todas).');
    }
    function rodarRota(\Illuminate\Routing\Route $r, array $params, \Illuminate\Http\Request $req): \Illuminate\Http\Response
    {
        K::$rotaAtual = $r;
        $req->rotaParams = $params;
        $mws = $r->mw;
        if (is_array($r->acao) && is_string($r->acao[0]) && class_exists($r->acao[0])) {
            $ctrl = make($r->acao[0]);
            $r->acao = [$ctrl, $r->acao[1]];
            if (method_exists($ctrl, 'middlewareDoController')) $mws = array_merge($mws, $ctrl->middlewareDoController($r->acao[1]));
            if (method_exists($ctrl, 'middleware') && is_callable([$ctrl, 'middleware']) && in_array(\Illuminate\Routing\Controllers\HasMiddleware::class, class_implements($ctrl), true)) {
                foreach ($ctrl::middleware() as $m) $mws[] = is_object($m) ? $m->middleware : $m;
            }
        }
        $fim = fn ($req) => paraResposta(chamar($r->acao, $params));
        foreach (array_reverse(array_unique(array_filter($mws, 'is_string'))) as $m) {
            $prox = $fim;
            $fim = fn ($req) => middleware($m, $req, $prox);
        }
        return paraResposta($fim($req));
    }
    function middleware(string $m, $req, \Closure $prox)
    {
        [$nome, $args] = array_pad(explode(':', $m, 2), 2, '');
        $args = $args === '' ? [] : explode(',', $args);
        if ($nome === 'auth' || $nome === 'auth.basic') {
            if (\Illuminate\Support\Facades\Auth::guest()) throw new \Illuminate\Auth\AuthenticationException('Unauthenticated.');
            return $prox($req);
        }
        if ($nome === 'guest') {
            if (\Illuminate\Support\Facades\Auth::check()) return redirect(\Illuminate\Support\Facades\Route::has('dashboard') ? route('dashboard') : '/');
            return $prox($req);
        }
        if (class_exists($nome) && method_exists($nome, 'handle')) {
            $r = make($nome)->handle($req, $prox, ...$args);
            return paraResposta($r);
        }
        return $prox($req);
    }

    function principal(array $req): array
    {
        $saida = '';
        $status = 200;
        $local = null;
        $tipo = 'text/html';
        try {
            iniciar($req);
            $modo = $req['modo'] ?? 'http';
            if ($modo === 'artisan') {
                ob_start();
                artisan(trim((string) ($req['cmd'] ?? 'list')));
                $saida = ob_get_clean();
                $tipo = 'text/plain';
            } elseif ($modo === 'previa') {
                K::$request = \Illuminate\Http\Request::criar('GET', '/', '');
                bancoSilencioso();
                $saida = embutirAssets(view($req['view'], $req['dados'] ?? [])->render());
            } else {
                K::$request = \Illuminate\Http\Request::criar($req['metodo'] ?? 'GET', $req['url'] ?? '/', $req['corpo'] ?? '', $req['cabecalhos'] ?? []);
                bancoAutomatico();
                providers();
                carregarRotas();
                ob_start();
                try {
                    $resp = despachar(K::$request);
                } catch (\Throwable $e) {
                    if ($e instanceof Parar) throw $e;
                    $resp = paginaDeErro($e);
                }
                $eco = ob_get_clean();
                $status = $resp->status;
                $local = $resp->local;
                $tipo = $resp->tipo;
                $saida = $eco . $resp->conteudo;
                if ($tipo === 'text/html' && !$local) $saida = embutirAssets($saida);
                if (K::$request->method() === 'GET' && $status === 200 && !$local && $tipo === 'text/html' && !K::$request->expectsJson()) K::$sessao->put('_previous.url', K::$request->fullUrl());
            }
        } catch (Parar $e) {
            $saida = $e->getMessage();
            while (ob_get_level() > ($GLOBALS['__wf_ob'] ?? 0)) $saida = ob_get_clean() . $saida;
        } catch (\Throwable $e) {
            while (ob_get_level() > ($GLOBALS['__wf_ob'] ?? 0)) ob_end_clean();
            $status = 500;
            if (($req['modo'] ?? 'http') === 'artisan') {
                [$arq, $linha] = localDoErro($e);
                K::$meta['erro'] = ['classe' => get_class($e), 'msg' => $e->getMessage(), 'arq' => $arq, 'linha' => $linha];
                $saida = get_class($e) . ': ' . $e->getMessage() . "\n  em " . $arq . ($linha ? ':' . $linha : '') . "\n";
                $tipo = 'text/plain';
            } else {
                $resp = paginaDeErro($e);
                $saida = $resp->conteudo;
                $tipo = $resp->tipo;
            }
        }
        while (ob_get_level() > 0 && ob_get_level() > ($GLOBALS['__wf_ob'] ?? 0)) $saida = ob_get_clean() . $saida;
        return [
            'status' => $status, 'local' => $local, 'tipo' => $tipo, 'corpo' => $saida,
            'sessao' => K::$sessao ? K::$sessao->paraGuardar() : [], 'logs' => K::$logs, 'erro' => K::$meta['erro'] ?? null,
            'rota' => K::$rotaAtual ? (K::$rotaAtual->nome ?: K::$rotaAtual->uri) : null,
            'usuario' => K::$sessao ? K::$sessao->get('login_web') : null,
        ];
    }
    function bancoSilencioso(): void
    {
        try { bancoAutomatico(); } catch (\Throwable $e) { K::log('warn', 'Não consegui preparar o banco: ' . $e->getMessage()); }
    }

    /* ---------- artisan ---------- */
    function artisan(string $cmd): void
    {
        $cmd = preg_replace('/^(php\s+)?artisan\s+/', '', $cmd);
        [$nome, $resto] = array_pad(preg_split('/\s+/', $cmd, 2), 2, '');
        $opc = fn ($o) => (bool) preg_match('/(^|\s)' . preg_quote($o, '/') . '(\s|=|$)/', $resto);
        switch ($nome) {
            case 'route:list':
                carregarRotas();
                $ls = [];
                foreach (K::$rotas as $r) {
                    $acao = $r->acao instanceof \Closure ? 'Closure' : (is_array($r->acao) ? (is_object($r->acao[0]) ? get_class($r->acao[0]) : $r->acao[0]) . '@' . $r->acao[1] : (string) $r->acao);
                    $ls[] = [implode('|', array_diff($r->metodos, ['HEAD'])), '/' . ltrim($r->uri, '/'), $r->nome ?? '', str_replace('App\\Http\\Controllers\\', '', $acao)];
                }
                if (!$ls) { echo "Nenhuma rota. Crie routes/web.php.\n"; return; }
                $w = [0, 0, 0];
                foreach ($ls as $l) for ($i = 0; $i < 3; $i++) $w[$i] = max($w[$i], mb_strlen($l[$i]));
                foreach ($ls as $l) echo str_pad($l[0], $w[0] + 2) . str_pad($l[1], $w[1] + 2) . str_pad($l[2], $w[2] + 2) . $l[3] . "\n";
                echo "\n  " . count($ls) . " rota(s)\n";
                return;
            case 'migrate':
                if ($opc('--fresh')) apagarTudo();
                $r = migrar();
                echo $r ? implode('', array_map(fn ($m) => "  ✓ $m\n", $r)) : "  Nada para migrar.\n";
                if ($opc('--seed') && semear()) echo "  ✓ DatabaseSeeder\n";
                return;
            case 'migrate:fresh':
                apagarTudo();
                echo "  Tabelas apagadas.\n";
                foreach (migrar() as $m) echo "  ✓ $m\n";
                if ($opc('--seed')) echo semear() ? "  ✓ DatabaseSeeder\n" : "  (sem database/seeders/DatabaseSeeder.php)\n";
                return;
            case 'migrate:refresh':
                reverter(1000);
                foreach (migrar() as $m) echo "  ✓ $m\n";
                if ($opc('--seed') && semear()) echo "  ✓ DatabaseSeeder\n";
                return;
            case 'migrate:rollback':
                $r = reverter(preg_match('/--step[= ](\d+)/', $resto, $m) ? (int) $m[1] : 1);
                echo $r ? implode('', array_map(fn ($m) => "  ↶ $m\n", $r)) : "  Nada para reverter.\n";
                return;
            case 'migrate:reset':
                foreach (reverter(1000) as $m) echo "  ↶ $m\n";
                return;
            case 'migrate:status':
                garantirMigrations();
                $feitas = q('select migration, batch from migrations')->fetchAll(\PDO::FETCH_KEY_PAIR);
                foreach (migrationsArqs() as $a) { $n = basename($a, '.php'); echo (isset($feitas[$n]) ? '  [' . $feitas[$n] . '] Ran     ' : '  Pending     ') . $n . "\n"; }
                return;
            case 'db:seed':
                $c = preg_match('/--class[= ]([\w\\\\]+)/', $resto, $m) ? $m[1] : 'Database\\Seeders\\DatabaseSeeder';
                if (!str_contains($c, '\\')) $c = 'Database\\Seeders\\' . $c;
                echo semear($c) ? "  ✓ $c\n" : "  Classe $c não encontrada.\n";
                return;
            case 'db:wipe':
                apagarTudo();
                echo "  Tabelas apagadas.\n";
                return;
            case 'tinker':
                $codigo = trim($resto);
                if ($codigo === '') { echo "Use: tinker App\\Models\\Post::count()\n"; return; }
                bancoAutomatico();
                providers();
                carregarRotas();
                $codigo = rtrim($codigo, '; ');
                $v = eval(preg_match('/^(echo|print|return|\$\w+\s*=|if|foreach|for|while)\b/', $codigo) ? $codigo . ';' : 'return ' . $codigo . ';');
                if ($v !== null) echo '= ' . mostrar($v) . "\n";
                return;
            case 'about':
                echo "  Laravel do navegador (Work Flower)\n  PHP " . PHP_VERSION . "\n  Banco: SQLite (" . (K::$req['db'] ?? ':memory:') . ")\n  Extensões: " . implode(', ', get_loaded_extensions()) . "\n";
                return;
            case 'wf:esquema':
                $ts = [];
                foreach (q("select name from sqlite_master where type = 'table' and name not like 'sqlite_%' order by name")->fetchAll(\PDO::FETCH_COLUMN) as $t) {
                    $cols = array_map(fn ($c) => $c['name'] . ($c['type'] ? ' ' . strtolower($c['type']) : '') . ($c['pk'] ? ' 🔑' : ''), q('PRAGMA table_info("' . $t . '")')->fetchAll());
                    $ts[] = ['nome' => $t, 'colunas' => $cols, 'total' => (int) q('select count(*) from "' . $t . '"')->fetchColumn()];
                }
                echo json_encode($ts, JSON_UNESCAPED_UNICODE);
                return;
            case 'wf:tabela':
                $t = trim($resto);
                $st = q('select * from "' . str_replace('"', '""', $t) . '" limit 200');
                $linhas = $st->fetchAll(\PDO::FETCH_NUM);
                $cols = [];
                for ($i = 0; $i < $st->columnCount(); $i++) $cols[] = $st->getColumnMeta($i)['name'];
                if (!$cols) $cols = array_column(q('PRAGMA table_info("' . str_replace('"', '""', $t) . '")')->fetchAll(), 'name');
                echo json_encode(['colunas' => $cols, 'linhas' => $linhas, 'total' => count($linhas), 'titulo' => $t], JSON_UNESCAPED_UNICODE);
                return;
            case 'list': case '':
                echo "Comandos do artisan no navegador:\n  route:list\n  migrate [--fresh] [--seed]\n  migrate:fresh [--seed]\n  migrate:rollback [--step=N]\n  migrate:refresh [--seed]\n  migrate:reset\n  migrate:status\n  db:seed [--class=NomeSeeder]\n  db:wipe\n  tinker <código PHP>   ex.: tinker App\\Models\\Post::count()\n  about\n\nmake:* não existe aqui: use o botão “+ Laravel” para criar controller, model, migration…\n";
                return;
            default:
                if (str_starts_with($nome, 'make:')) { echo "Aqui não há terminal: use o botão “+ Laravel” para criar arquivos.\n"; return; }
                echo "Comando “$nome” não existe no artisan do navegador. Digite list para ver os disponíveis.\n";
        }
    }
    function mostrar($v): string
    {
        if ($v instanceof \Illuminate\Contracts\Support\Arrayable) $v = $v->toArray();
        if (is_array($v) || is_object($v)) return json_encode($v, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES) ?: print_r($v, true);
        return var_export($v, true);
    }

    class Arr
    {
        public static function get($a, $k, $d = null)
        {
            if ($k === null) return $a;
            if (is_array($a) && array_key_exists($k, $a)) return $a[$k];
            foreach (explode('.', (string) $k) as $p) {
                if (is_array($a) && array_key_exists($p, $a)) $a = $a[$p];
                elseif ($a instanceof \ArrayAccess && isset($a[$p])) $a = $a[$p];
                elseif (is_object($a) && isset($a->{$p})) $a = $a->{$p};
                else return $d instanceof \Closure ? $d() : $d;
            }
            return $a;
        }
        public static function set(&$a, $k, $v)
        {
            $ps = explode('.', $k);
            $c = &$a;
            foreach ($ps as $p) { if (!isset($c[$p]) || !is_array($c[$p])) $c[$p] = []; $c = &$c[$p]; }
            $c = $v;
            return $a;
        }
        public static function has($a, $k): bool { return static::get($a, $k, "\0nada") !== "\0nada"; }
        public static function only($a, $ks) { return array_intersect_key($a, array_flip((array) $ks)); }
        public static function except($a, $ks) { return array_diff_key($a, array_flip((array) $ks)); }
        public static function first($a, $cb = null, $d = null) { foreach ($a as $k => $v) if (!$cb || $cb($v, $k)) return $v; return $d; }
        public static function last($a, $cb = null, $d = null) { return static::first(array_reverse($a, true), $cb, $d); }
        public static function pluck($a, $v, $k = null) { return (new Collection($a))->pluck($v, $k)->all(); }
        public static function wrap($v) { return is_null($v) ? [] : (is_array($v) ? $v : [$v]); }
        public static function flatten($a, $prof = INF) { $r = []; foreach ($a as $v) { if (is_array($v) && $prof > 0) foreach (static::flatten($v, $prof - 1) as $x) $r[] = $x; else $r[] = $v; } return $r; }
        public static function random($a, $n = null) { $k = array_rand($a, $n ?: 1); return $n ? array_map(fn ($i) => $a[$i], (array) $k) : $a[$k]; }
        public static function isAssoc($a) { return array_keys($a) !== range(0, count($a) - 1); }

    class Route
    {
        public array $metodos;
        public string $uri;
        public $acao;
        public ?string $nome = null;
        public string $prefixoNome = '';
        public array $wheres = [];
        public array $mw = [];

        public function __construct(array $metodos, string $uri, $acao, array $mw, string $prefixoNome)
        {
            $this->metodos = $metodos;
            $this->uri = $uri;
            $this->acao = $acao;
            $this->mw = $mw;
            $this->prefixoNome = $prefixoNome;
        }
        public function name($n) { $this->nome = $this->prefixoNome . $n; return $this; }
        public function middleware(...$m) { foreach ($m as $x) foreach ((array) $x as $y) $this->mw[] = $y; return $this; }
        public function withoutMiddleware(...$m) { $tirar = array_merge(...array_map(fn ($x) => (array) $x, $m)); $this->mw = array_values(array_diff($this->mw, $tirar)); return $this; }
        public function where($k, $v = null) { foreach (is_array($k) ? $k : [$k => $v] as $a => $b) $this->wheres[$a] = $b; return $this; }
        public function whereNumber($k) { foreach ((array) $k as $a) $this->wheres[$a] = '[0-9]+'; return $this; }
        public function whereAlpha($k) { foreach ((array) $k as $a) $this->wheres[$a] = '[a-zA-Z]+'; return $this; }
        public function whereAlphaNumeric($k) { foreach ((array) $k as $a) $this->wheres[$a] = '[a-zA-Z0-9]+'; return $this; }
        public function whereUuid($k) { foreach ((array) $k as $a) $this->wheres[$a] = '[\da-fA-F]{8}-[\da-fA-F]{4}-[\da-fA-F]{4}-[\da-fA-F]{4}-[\da-fA-F]{12}'; return $this; }
        public function whereIn($k, array $vs) { foreach ((array) $k as $a) $this->wheres[$a] = implode('|', array_map(fn ($v) => preg_quote((string) $v, '#'), $vs)); return $this; }
        public function getName() { return $this->nome; }
        public function uri() { return $this->uri; }
        public function parameters() { return K::$request ? K::$request->rotaParams : []; }
        public function parameter($k, $d = null) { return $this->parameters()[$k] ?? $d; }
        public function named(...$p) { return Str::is($p, (string) $this->nome); }
        public function __call($m, $a) { return $this; }

        public function regex(): string
        {
            $uri = trim($this->uri, '/');
            if ($uri === '') return '#^/?$#';
            $re = '';
            foreach (explode('/', $uri) as $seg) {
                if (preg_match('/^\{(\w+)(\?)?\}$/', $seg, $m)) {
                    $pat = $this->wheres[$m[1]] ?? '[^/]+';
                    $re .= !empty($m[2]) ? '(?:/(?P<' . $m[1] . '>' . $pat . '))?' : '/(?P<' . $m[1] . '>' . $pat . ')';
                    continue;
                }
                $re .= '/' . preg_replace_callback('/\\\\\{(\w+)(\\\\\?)?\\\\\}/', fn ($m) => '(?P<' . $m[1] . '>' . ($this->wheres[$m[1]] ?? '[^/]+') . ')' . (!empty($m[2]) ? '?' : ''), preg_quote($seg, '#'));
            }
            return '#^' . $re . '/?$#u';
        }
        public function casar(string $caminho): ?array
        {
            if (!preg_match($this->regex(), $caminho, $m)) return null;
            $p = [];
            foreach ($m as $k => $v) if (is_string($k) && $v !== '') $p[$k] = rawurldecode($v);
            return $p;
        }
        public function nomesParams(): array { preg_match_all('/\{(\w+)\??\}/', $this->uri, $m); return $m[1]; }
        public function gerar($params): string
        {
            $params = is_array($params) ? $params : [$params];
            $chave = fn ($v) => $v instanceof \Illuminate\Database\Eloquent\Model ? $v->getRouteKey() : ($v instanceof \BackedEnum ? $v->value : $v);
            $uri = preg_replace_callback('#\{(\w+)(\?)?\}#', function ($m) use (&$params, $chave) {
                if (array_key_exists($m[1], $params)) { $v = $params[$m[1]]; unset($params[$m[1]]); return rawurlencode((string) $chave($v)); }
                foreach ($params as $k => $v) if (is_int($k)) { unset($params[$k]); return rawurlencode((string) $chave($v)); }
                if (!empty($m[2])) return "\0";
                throw new \Illuminate\Routing\Exceptions\UrlGenerationException('Missing required parameter for [Route: ' . $this->nome . '] [URI: ' . $this->uri . '] [Missing parameter: ' . $m[1] . '].');
            }, $this->uri);
            $uri = str_replace(["/\0", "\0"], '', $uri);
            $resto = array_filter($params, fn ($k) => is_string($k), ARRAY_FILTER_USE_KEY);
            return '/' . ltrim($uri, '/') . ($resto ? '?' . http_build_query(array_map($chave, $resto)) : '');
        }
    }

    class Router
    {
        public static function grupo(): array
        {
            $g = ['prefix' => '', 'name' => '', 'mw' => [], 'controller' => null, 'where' => []];
            foreach (K::$pilha as $f) {
                $g['prefix'] = trim($g['prefix'] . '/' . trim($f['prefix'] ?? '', '/'), '/');
                $g['name'] .= $f['name'] ?? '';
                $g['mw'] = array_merge($g['mw'], $f['mw'] ?? []);
                $g['where'] = array_merge($g['where'], $f['where'] ?? []);
                if (!empty($f['controller'])) $g['controller'] = $f['controller'];
            }
            return $g;
        }
        public static function adicionar(array $metodos, string $uri, $acao, ?array $g = null): Route
        {
            $g ??= static::grupo();
            $uri = trim(trim($g['prefix'], '/') . '/' . trim($uri, '/'), '/');
            if (is_string($acao) && str_contains($acao, '@')) {
                [$c, $m] = explode('@', $acao, 2);
                if (!str_contains($c, '\\')) $c = 'App\\Http\\Controllers\\' . $c;
                $acao = [$c, $m];
            } elseif (is_string($acao) && $g['controller']) {
                $acao = [$g['controller'], $acao];
            } elseif (is_string($acao)) {
                $acao = [$acao, '__invoke'];
            } elseif (is_array($acao) && count($acao) === 1) {
                $acao = [$acao[0], '__invoke'];
            } elseif ($acao === null) {
                $acao = fn () => '';
            }
            $r = new Route($metodos, $uri, $acao, $g['mw'], $g['name']);
            $r->wheres = $g['where'];
            K::$rotas[] = $r;
            return $r;
        }
        public static function porNome(string $n): ?Route
        {
            foreach (K::$rotas as $r) if ($r->nome === $n) return $r;
            return null;
        }
    }

    class PendingResourceRegistration
    {
        public ?array $g;
        protected array $acoes;
        protected array $nomes = [];
        protected array $params = [];
        protected array $mw = [];
        protected bool $feito = false;
        public function __construct(protected string $nome, protected string $ctrl, array $o, bool $api)
        {
            $this->g = Router::grupo();
            $this->acoes = $api ? ['index', 'store', 'show', 'update', 'destroy'] : ['index', 'create', 'store', 'show', 'edit', 'update', 'destroy'];
            if (isset($o['only'])) $this->only($o['only']);
            if (isset($o['except'])) $this->except($o['except']);
            if (isset($o['names'])) $this->names($o['names']);
            if (isset($o['parameters'])) $this->parameters($o['parameters']);
        }
        public function only($a) { $a = is_array($a) ? $a : func_get_args(); $this->acoes = array_values(array_intersect($this->acoes, $a)); return $this; }
        public function except($a) { $a = is_array($a) ? $a : func_get_args(); $this->acoes = array_values(array_diff($this->acoes, $a)); return $this; }
        public function names($n) { if (is_string($n)) { foreach (['index', 'create', 'store', 'show', 'edit', 'update', 'destroy'] as $a) $this->nomes[$a] = $n . '.' . $a; } else $this->nomes = $n + $this->nomes; return $this; }
        public function name($a, $n) { $this->nomes[$a] = $n; return $this; }
        public function parameters($p) { $this->params = $p; return $this; }
        public function parameter($k, $v) { $this->params[$k] = $v; return $this; }
        public function middleware(...$m) { foreach ($m as $x) foreach ((array) $x as $y) $this->mw[] = $y; return $this; }
        public function shallow() { return $this; }
        public function scoped($x = []) { return $this; }
        public function withTrashed($x = []) { return $this; }
        public function missing($cb) { return $this; }
        public function registrar()
        {
            if ($this->feito) return;
            $this->feito = true;
            $partes = explode('.', $this->nome);
            $base = '';
            foreach ($partes as $i => $p) {
                if ($i < count($partes) - 1) { $base .= $p . '/{' . ($this->params[$p] ?? Str::singular(str_replace('-', '_', $p))) . '}/'; continue; }
                $base .= $p;
            }
            $ultimo = end($partes);
            $param = $this->params[$ultimo] ?? Str::singular(str_replace('-', '_', $ultimo));
            $def = ['index' => [['GET', 'HEAD'], ''], 'create' => [['GET', 'HEAD'], '/create'], 'store' => [['POST'], ''], 'show' => [['GET', 'HEAD'], '/{' . $param . '}'],
                'edit' => [['GET', 'HEAD'], '/{' . $param . '}/edit'], 'update' => [['PUT', 'PATCH'], '/{' . $param . '}'], 'destroy' => [['DELETE'], '/{' . $param . '}']];
            foreach (['index', 'create', 'store', 'show', 'edit', 'update', 'destroy'] as $a) {
                if (!in_array($a, $this->acoes, true)) continue;
                $r = Router::adicionar($def[$a][0], $base . $def[$a][1], [$this->ctrl, $a], $this->g);
                $r->mw = array_merge($r->mw, $this->mw);
                $r->name($this->nomes[$a] ?? (str_replace('/', '.', $this->nome) . '.' . $a));
            }
        }
        public function __destruct() { $this->registrar(); }
    }

    class Request implements \ArrayAccess
    {
        public string $metodo = 'GET';
        public string $caminho = '/';
        public array $consulta = [];
        public array $corpo = [];
        public array $cabecalhos = [];
        public array $rotaParams = [];
        public array $atributos = [];
        public $validador = null;

        public static function criar(string $metodo, string $url, string $corpo = '', array $cab = [])
        {
            $r = new static();
            $u = parse_url($url) ?: [];
            $r->caminho = '/' . trim(rawurldecode($u['path'] ?? '/'), '/');
            parse_str($u['query'] ?? '', $r->consulta);
            parse_str($corpo, $r->corpo);
            if ($corpo !== '' && !$r->corpo && in_array(substr(ltrim($corpo), 0, 1), ['{', '['], true)) $r->corpo = (array) json_decode($corpo, true);
            $r->cabecalhos = array_change_key_case($cab, CASE_LOWER);
            // como os middlewares TrimStrings e ConvertEmptyStringsToNull do Laravel: "  x " vira "x" e "" vira null
            $limpar = function ($v, $k = '') use (&$limpar) {
                if (is_array($v)) { foreach ($v as $a => $b) $v[$a] = $limpar($b, $a); return $v; }
                if (!is_string($v) || in_array($k, ['password', 'password_confirmation', 'current_password'], true)) return $v;
                $t = preg_replace('~^[\s\x{FEFF}\x{200B}]+|[\s\x{FEFF}\x{200B}]+$~u', '', $v) ?? trim($v);
                return $t === '' ? null : $t;
            };
            $r->consulta = $limpar($r->consulta);
            $r->corpo = $limpar($r->corpo);
            $metodo = strtoupper($metodo);
            if ($metodo === 'POST' && isset($r->corpo['_method'])) $metodo = strtoupper((string) $r->corpo['_method']);
            $r->metodo = $metodo;
            return $r;

    class Validator
    {
        protected MessageBag $erros;
        protected array $regras = [];
        protected bool $rodou = false;
        protected array $depois = [];
        public function __construct(protected array $dados, array $regras, protected array $msgs = [], protected array $nomes = [])
        {
            $this->erros = new MessageBag();
            foreach ($regras as $campo => $r) {
                $lista = is_string($r) ? explode('|', $r) : (is_array($r) ? $r : [$r]);
                $exp = [];
                foreach ($lista as $x) { if (is_string($x) && str_contains($x, '|')) foreach (explode('|', $x) as $y) $exp[] = $y; else $exp[] = $x; }
                if (str_contains($campo, '*')) {
                    $re = '#^' . str_replace('\*', '[^.]+', preg_quote($campo, '#')) . '$#';
                    foreach (array_keys(static::achatar($dados)) as $k) if (preg_match($re, $k)) $this->regras[$k] = $exp;
                    continue;
                }
                $this->regras[$campo] = $exp;
            }
        }
        protected static function achatar($a, $p = '') { $r = []; foreach ((array) $a as $k => $v) { $r[$p . $k] = $v; if (is_array($v)) $r += static::achatar($v, $p . $k . '.'); } return $r; }
        public function after($cb) { $this->depois[] = $cb; return $this; }
        public function sometimes($c, $r, $cb) { if ($cb(new \Illuminate\Support\Fluent($this->dados))) foreach ((array) $c as $x) $this->regras[$x] = array_merge($this->regras[$x] ?? [], is_string($r) ? explode('|', $r) : $r); return $this; }
        public function errors() { $this->rodar(); return $this->erros; }
        public function messages() { return $this->errors(); }
        public function fails() { return $this->errors()->any(); }
        public function passes() { return !$this->fails(); }
        public function failed() { return $this->errors()->messages(); }
        public function validate() { if ($this->fails()) throw new ValidationException($this); return $this->validated(); }
        public function validated()
        {
            if ($this->fails()) throw new ValidationException($this);
            $r = [];
            foreach ($this->regras as $k => $_) if (Arr::has($this->dados, $k)) Arr::set($r, $k, Arr::get($this->dados, $k));
            return $r;
        }
        public function safe() { return new \Illuminate\Support\Fluent($this->validated()); }
        public function getData() { return $this->dados; }
        public function setData(array $d) { $this->dados = $d; $this->rodou = false; return $this; }

        protected function nome($c) { return $this->nomes[$c] ?? str_replace('_', ' ', preg_replace('/\.\d+/', '', $c)); }
        protected function vazio($v) { return $v === null || (is_string($v) && trim($v) === '') || (is_array($v) && !$v); }
        protected function tipoTamanho(array $rs, $v) { foreach ($rs as $r) if (in_array($r, ['numeric', 'integer', 'decimal'], true)) return 'numeric'; return is_array($v) ? 'array' : 'string'; }
        protected function tamanho(array $rs, $v) { $t = $this->tipoTamanho($rs, $v); return $t === 'numeric' ? (float) $v : ($t === 'array' ? count($v) : mb_strlen((string) $v)); }
        protected function rodar()
        {
            if ($this->rodou) return;
            $this->rodou = true;
            foreach ($this->regras as $campo => $rs) {
                $v = Arr::get($this->dados, $campo);
                $nomes = array_map(fn ($r) => is_string($r) ? explode(':', $r, 2)[0] : '', $rs);
                if (in_array('sometimes', $nomes, true) && !Arr::has($this->dados, $campo)) continue;
                $anulavel = in_array('nullable', $nomes, true);
                $exigido = in_array('required', $nomes, true) || in_array('accepted', $nomes, true);
                foreach ($rs as $r) {
                    if ($r instanceof \Closure) {
                        if ($this->vazio($v) && !$exigido) continue;
                        $r($campo, $v, function ($m) use ($campo) { $this->erros->add($campo, str_replace(':attribute', $this->nome($campo), $m)); return new \WF\Nada(); });
                        continue;
                    }
                    if (is_object($r)) {
                        if ($r instanceof Rules\Regra) { if (!$r->passa($campo, $v, $this)) $this->falhar($campo, $r->mensagemPadrao($this->nome($campo)), $r->chave(), []); continue; }
                        if (method_exists($r, 'validate')) { if ($this->vazio($v) && !$exigido) continue; $r->validate($campo, $v, function ($m) use ($campo) { $this->erros->add($campo, str_replace(':attribute', $this->nome($campo), $m)); return new \WF\Nada(); }); continue; }
                        if (method_exists($r, 'passes')) { if (!$r->passes($campo, $v)) $this->erros->add($campo, str_replace(':attribute', $this->nome($campo), (string) $r->message())); continue; }
                        if (method_exists($r, '__toString')) $r = (string) $r; else continue;
                    }
                    [$nome, $p] = array_pad(explode(':', $r, 2), 2, '');
                    $args = $p === '' ? [] : str_getcsv($p);
                    if ($nome === 'required_if') { if (count($args) >= 2 && in_array((string) Arr::get($this->dados, $args[0]), array_slice($args, 1))) $nome = 'required'; else continue; }
                    if ($nome === 'required_unless') { if (count($args) >= 2 && !in_array((string) Arr::get($this->dados, $args[0]), array_slice($args, 1))) $nome = 'required'; else continue; }
                    if ($nome === 'required_with') { $ok = false; foreach ($args as $a) if (!$this->vazio(Arr::get($this->dados, $a))) $ok = true; if ($ok) $nome = 'required'; else continue; }
                    if ($nome === 'required_without') { $ok = false; foreach ($args as $a) if ($this->vazio(Arr::get($this->dados, $a))) $ok = true; if ($ok) $nome = 'required'; else continue; }
                    if ($nome !== 'required' && $nome !== 'accepted' && $nome !== 'present' && $nome !== 'confirmed' && $this->vazio($v)) continue;
                    if ($anulavel && $v === null) continue;
                    if (!$this->testar($nome, $args, $v, $campo, $rs)) { $this->falhar($campo, null, $nome, $args, $rs, $v); if ($nome === 'required') break; }
                }
            }
            foreach ($this->depois as $cb) $cb($this);
        }
        protected function testar($n, $a, $v, $campo, $rs)
        {
            switch ($n) {
                case 'required': return !$this->vazio($v);
                case 'present': return Arr::has($this->dados, $campo);
                case 'filled': return !$this->vazio($v);
                case 'accepted': return in_array($v, ['yes', 'on', '1', 1, true, 'true'], true);
                case 'declined': return in_array($v, ['no', 'off', '0', 0, false, 'false'], true);
                case 'string': return is_string($v);
                case 'integer': case 'int': return filter_var($v, FILTER_VALIDATE_INT) !== false;
                case 'numeric': return is_numeric($v);
                case 'decimal': return is_numeric($v);
                case 'boolean': case 'bool': return in_array($v, [true, false, 0, 1, '0', '1'], true);
                case 'array': return is_array($v);
                case 'list': return is_array($v) && array_is_list($v);
                case 'email': return (bool) filter_var($v, FILTER_VALIDATE_EMAIL);
                case 'url': case 'active_url': return (bool) filter_var($v, FILTER_VALIDATE_URL);
                case 'uuid': return (bool) preg_match('/^[\da-f]{8}-[\da-f]{4}-[\da-f]{4}-[\da-f]{4}-[\da-f]{12}$/i', (string) $v);
                case 'ip': return (bool) filter_var($v, FILTER_VALIDATE_IP);
                case 'json': json_decode((string) $v); return json_last_error() === JSON_ERROR_NONE;
                case 'alpha': return (bool) preg_match('/^[\pL\pM]+$/u', (string) $v);
                case 'alpha_num': return (bool) preg_match('/^[\pL\pM\pN]+$/u', (string) $v);
                case 'alpha_dash': return (bool) preg_match('/^[\pL\pM\pN_-]+$/u', (string) $v);
                case 'lowercase': return mb_strtolower((string) $v) === (string) $v;
                case 'uppercase': return mb_strtoupper((string) $v) === (string) $v;
                case 'date': return strtotime((string) $v) !== false;
                case 'date_format': $d = \DateTime::createFromFormat($a[0], (string) $v); return $d && $d->format($a[0]) === (string) $v;
                case 'after': case 'before': case 'after_or_equal': case 'before_or_equal':
                    $o = Arr::get($this->dados, $a[0]) ?? $a[0];
                    $x = strtotime((string) $v); $y = strtotime((string) $o);
                    if ($x === false || $y === false) return false;
                    return match ($n) { 'after' => $x > $y, 'before' => $x < $y, 'after_or_equal' => $x >= $y, default => $x <= $y };
                case 'min': return $this->tamanho($rs, $v) >= (float) $a[0];
                case 'max': return $this->tamanho($rs, $v) <= (float) $a[0];
                case 'between': $t = $this->tamanho($rs, $v); return $t >= (float) $a[0] && $t <= (float) $a[1];
                case 'size': return $this->tamanho($rs, $v) == (float) $a[0];
                case 'gt': case 'gte': case 'lt': case 'lte':
                    $o = is_numeric($a[0]) ? (float) $a[0] : (float) Arr::get($this->dados, $a[0]);
                    $t = $this->tamanho($rs, $v);
                    return match ($n) { 'gt' => $t > $o, 'gte' => $t >= $o, 'lt' => $t < $o, default => $t <= $o };
                case 'digits': return ctype_digit((string) $v) && strlen((string) $v) == $a[0];
                case 'digits_between': return ctype_digit((string) $v) && strlen((string) $v) >= $a[0] && strlen((string) $v) <= $a[1];
                case 'in': return in_array((string) (is_bool($v) ? (int) $v : $v), $a, true);
                case 'not_in': return !in_array((string) $v, $a, true);
                case 'confirmed': return $v === Arr::get($this->dados, $campo . '_confirmation');
                case 'same': return $v === Arr::get($this->dados, $a[0]);
                case 'different': return $v !== Arr::get($this->dados, $a[0]);
                case 'regex': return (bool) preg_match($a ? implode(',', $a) : '//', (string) $v);
                case 'not_regex': return !preg_match(implode(',', $a), (string) $v);
                case 'starts_with': foreach ($a as $x) if (str_starts_with((string) $v, $x)) return true; return false;
                case 'ends_with': foreach ($a as $x) if (str_ends_with((string) $v, $x)) return true; return false;
                case 'unique':
                    $t = (new Rules\Unique($a[0] ?? '', $a[1] ?? 'NULL'));
                    if (isset($a[2]) && $a[2] !== '' && strtoupper($a[2]) !== 'NULL') $t->ignore($a[2], $a[3] ?? 'id');
                    return $t->passa($campo, $v, $this);
                case 'exists': return (new Rules\Exists($a[0] ?? '', $a[1] ?? 'NULL'))->passa($campo, $v, $this);
                case 'current_password': $u = \Illuminate\Support\Facades\Auth::user(); return $u && \Illuminate\Support\Facades\Hash::check($v, $u->getAttribute('password'));
                case 'password': return mb_strlen((string) $v) >= 8;
                case 'image': case 'file': case 'mimes': case 'mimetypes': case 'dimensions': case 'extensions': return true;
                default: return true;   // sometimes, nullable, bail, regras desconhecidas
            }
        }

    class Unique extends Regra
    {
        protected $ignorar = null;
        protected $colIgnorar = 'id';
        protected array $wheres = [];
        public function __construct(protected $tabela, protected $coluna = 'NULL')
        {
            if (class_exists($tabela) && is_subclass_of($tabela, \Illuminate\Database\Eloquent\Model::class)) $this->tabela = (new $tabela())->getTable();
        }
        public function ignore($id, $col = null) { if ($id instanceof \Illuminate\Database\Eloquent\Model) { $col ??= $id->getKeyName(); $id = $id->getKey(); } $this->ignorar = $id; $this->colIgnorar = $col ?? 'id'; return $this; }
        public function ignoreModel($m, $col = null) { return $this->ignore($m, $col); }
        public function where($c, $v = null) { $this->wheres[] = [$c, $v]; return $this; }
        public function whereNull($c) { $this->wheres[] = [$c, null]; return $this; }
        public function withoutTrashed() { return $this->whereNull('deleted_at'); }
        public function chave() { return 'unique'; }
        public function passa($campo, $v, $validador)
        {
            $col = strtoupper((string) $this->coluna) === 'NULL' ? preg_replace('/\.\d+$/', '', $campo) : $this->coluna;
            $q = \Illuminate\Support\Facades\DB::table($this->tabela)->where($col, $v);
            if ($this->ignorar !== null && $this->ignorar !== '') $q->where($this->colIgnorar, '!=', $this->ignorar);
            foreach ($this->wheres as [$c, $x]) $x === null ? $q->whereNull($c) : $q->where($c, $x);
            return !$q->exists();
        }
    }
    class Exists extends Unique
    {
        public function chave() { return 'exists'; }
        public function passa($campo, $v, $validador)
        {
            $col = strtoupper((string) $this->coluna) === 'NULL' ? preg_replace('/\.\d+$/', '', $campo) : $this->coluna;
            $q = \Illuminate\Support\Facades\DB::table($this->tabela);
            is_array($v) ? $q->whereIn($col, $v) : $q->where($col, $v);
            foreach ($this->wheres as [$c, $x]) $x === null ? $q->whereNull($c) : $q->where($c, $x);
            return is_array($v) ? $q->count() >= count(array_unique($v)) : $q->exists();
        }
    }

    {
        protected int $min = 8;
        protected array $exige = [];
        public static function defaults($cb = null) { return new static(); }
        public static function min($n) { $p = new static(); $p->min = $n; return $p; }
        public function letters() { $this->exige[] = 'letras'; return $this; }

        public function numbers() { $this->exige[] = 'numeros'; return $this; }
        public function symbols() { $this->exige[] = 'simbolos'; return $this; }
        public function uncompromised($n = 0) { return $this; }

        {
            $v = (string) $v;
            if (mb_strlen($v) < $this->min) { $this->falhou = 'deve ter pelo menos ' . $this->min . ' caracteres'; return false; }
            if (in_array('letras', $this->exige) && !preg_match('/\pL/u', $v)) { $this->falhou = 'deve ter pelo menos uma letra'; return false; }
            if (in_array('maiusculas', $this->exige) && !(preg_match('/\p{Lu}/u', $v) && preg_match('/\p{Ll}/u', $v))) { $this->falhou = 'deve ter letras maiúsculas e minúsculas'; return false; }

            return true;
        }
        public function mensagemPadrao($nome) { return 'O campo ' . $nome . ' ' . $this->falhou . '.'; }

        public function compilarWheres(array $ws): array
        {
            $sql = '';
            $b = [];
            foreach ($ws as $i => $w) {
                $pre = $i === 0 ? '' : ' ' . $w['bool'] . ' ';
                switch ($w['tipo']) {
                    case 'basico':
                        if ($w['v'] instanceof Expression) { $sql .= $pre . $this->col($w['c']) . ' ' . $w['op'] . ' ' . $w['v']; break; }
                        $sql .= $pre . $this->col($w['c']) . ' ' . $w['op'] . ' ?';
                        $b[] = $w['v'];
                        break;
                    case 'in':
                        if (!$w['vs']) { $sql .= $pre . ($w['nao'] ? '1 = 1' : '0 = 1'); break; }
                        $sql .= $pre . $this->col($w['c']) . ($w['nao'] ? ' not in (' : ' in (') . implode(', ', array_fill(0, count($w['vs']), '?')) . ')';
                        array_push($b, ...$w['vs']);
                        break;
                    case 'nulo': $sql .= $pre . $this->col($w['c']) . ($w['nao'] ? ' is not null' : ' is null'); break;
                    case 'entre': $sql .= $pre . $this->col($w['c']) . ($w['nao'] ? ' not between ? and ?' : ' between ? and ?'); array_push($b, $w['vs'][0], $w['vs'][1]); break;
                    case 'raw': $sql .= $pre . $w['sql']; array_push($b, ...$w['b']); break;
                    case 'grupo': case 'nao':
                        [$s, $bb] = $w['q']->compilarWheres($w['q']->wheres);
                        $sql .= $pre . ($w['tipo'] === 'nao' ? 'not ' : '') . '(' . $s . ')';
                        array_push($b, ...$bb);
                        break;
                }
            }
            return [$sql, $b];
        }

    /** @vite: o CSS e o JS do projeto entram na página; Tailwind (@tailwind / @import "tailwindcss") vira o Tailwind do CDN. */
    function __vite(array $entradas)
    {
        $h = '';
        $tw = false;
        foreach ($entradas as $e) {
            $f = K::caminho($e);
            if (!is_file($f)) { $h .= '<!-- @vite: ' . e($e) . ' não existe no projeto -->'; continue; }
            $t = (string) file_get_contents($f);
            if (preg_match('/\.(css|scss|sass|less)$/i', $e)) {
                if (preg_match('/@tailwind\b|@import\s+["\']tailwindcss/', $t)) { $tw = true; $t = preg_replace('/@tailwind\b[^;]*;|@import\s+["\']tailwindcss[^;]*;|@source[^;]*;|@plugin[^;]*;|@theme\s*\{[^}]*\}/', '', $t); }
                $t = preg_replace('/@import\s+["\'][^"\']+["\'][^;]*;/', '', $t);
                if (trim($t) !== '') $h .= '<style>' . str_ireplace('</style', '<\/style', $t) . '</style>';
            } elseif (preg_match('/\.(js|mjs|ts)$/i', $e)) {
                if (preg_match('/^\s*import\s/m', $t)) { $h .= '<!-- @vite: ' . e($e) . ' usa import de pacotes do npm (axios, alpinejs…), que precisam do Vite; ficou de fora da prévia -->'; continue; }
                $h .= '<script type="module">' . str_ireplace('</script', '<\/script', $t) . '</script>';
            }
        }
        if ($tw) $h = '<script src="https://cdn.tailwindcss.com"></script>' . $h;
        return new \Illuminate\Support\HtmlString($h);
    }
